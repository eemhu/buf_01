112 public boolean hasNext() {
113 final boolean rv;
114 if (limit.get() == -1) {
115 // limit not set, ignore
116 rv = currentOffset.get() < origin.leasedObject().byteSize();
117 }
118 else {
119 rv = currentOffset.get() < Math.min(limit.get(), origin.leasedObject().byteSize());
120 }
121 return rv;
122 }
123
124 public byte next() {
125 if (!hasNext()) {
126 throw new IndexOutOfBoundsException("Reached end of segment or limit, cannot provide next byte");
127 }
128 final long nextIndex = currentOffset.getAndIncrement();
129
130 return origin.leasedObject().get(ValueLayout.JAVA_BYTE, nextIndex);
131 }
132
133 public void write(final byte b) {
134 if (!hasNext()) {
135 throw new IndexOutOfBoundsException("Reached end of segment or limit, cannot write to next byte");
136 }
137
138 final long nextIndex = currentOffset.getAndIncrement();
139
140 origin.leasedObject().set(ValueLayout.JAVA_BYTE, nextIndex, b);
141 }
142
143 public long currentPosition() {
144 return currentOffset.get();
145 }
146
147 public void position(final long newPosition) {
148 final long segmentByteSize = leasedObject().byteSize();
149 if (newPosition < 0 || newPosition > segmentByteSize) {
150 throw new IndexOutOfBoundsException(
151 "New position was out of bounds; expected value between 0 and " + segmentByteSize + "; was "
152 + newPosition
153 );
154 }
155 currentOffset.set(newPosition);
156 }
157
158 public long currentLimit() {
159 return limit.get();
160 }
161
162 public void limit(final long newLimit) {
163 final long segmentByteSize = leasedObject().byteSize();
164 if (newLimit < -1 || newLimit > segmentByteSize) {
165 throw new IndexOutOfBoundsException(
166 "New limit was out of bounds; expected value between -1 and " + segmentByteSize + "; was "
167 + newLimit
168 );
169 }
170
171 limit.set(newLimit);
172 }
