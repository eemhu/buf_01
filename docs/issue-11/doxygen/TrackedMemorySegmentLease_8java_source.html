111 public boolean hasNext() {
112 final boolean rv;
113 if (limit.get() == -1) {
114 // limit not set, ignore
115 rv = currentOffset.get() < origin.leasedObject().byteSize();
116 }
117 else {
118 rv = currentOffset.get() < Math.min(limit.get(), origin.leasedObject().byteSize());
119 }
120 return rv;
121 }
122
123 public byte next() {
124 if (!hasNext()) {
125 throw new IndexOutOfBoundsException("Reached end of segment or limit, cannot provide next byte");
126 }
127 final long nextIndex = currentOffset.getAndIncrement();
128
129 return origin.leasedObject().get(ValueLayout.JAVA_BYTE, nextIndex);
130 }
131
132 public void write(final byte b) {
133 if (!hasNext()) {
134 throw new IndexOutOfBoundsException("Reached end of segment or limit, cannot write to next byte");
135 }
136
137 final long nextIndex = currentOffset.getAndIncrement();
138
139 origin.leasedObject().set(ValueLayout.JAVA_BYTE, nextIndex, b);
140 }
141
142 public long currentPosition() {
143 return currentOffset.get();
144 }
145
146 public void position(final long newPosition) {
147 final long segmentByteSize = leasedObject().byteSize();
148 if (newPosition < 0 || newPosition > segmentByteSize) {
149 throw new IndexOutOfBoundsException(
150 "New position was out of bounds; expected value between 0 and " + segmentByteSize + "; was "
151 + newPosition
152 );
153 }
154 currentOffset.set(newPosition);
155 }
156
157 public long currentLimit() {

159 }

161 public void limit(final long newLimit) {
162 final long segmentByteSize = leasedObject().byteSize();
163 if (newLimit < -1 || newLimit > segmentByteSize) {
164 throw new IndexOutOfBoundsException(
165 "New limit was out of bounds; expected value between -1 and " + segmentByteSize + "; was "
166 + newLimit
167 );
168 }
169
170 limit.set(newLimit);
